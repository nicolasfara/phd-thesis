% Aggregate computing: a network of devices (left) whose single device repeats a
% sense-compute-interact round (right), a redraw of figures/ac.svg.
% Requires: \usepackage{tikz}
%   \usetikzlibrary{arrows.meta,positioning,fit,backgrounds}
% Include with a normal figure environment, e.g. \input{figures/ac}.
\begin{tikzpicture}[
  >=Latex, font=\small,
  dev/.style={circle,draw,fill=gray!50,minimum size=3.4mm,inner sep=0pt},
  hub/.style={dev,minimum size=4.6mm},
  idle/.style={dev,draw=gray!50,fill=gray!22},
  link/.style={draw,gray!75},
  faint/.style={draw,gray!40},
  msg/.style={->,gray!85,shorten >=1.3mm,shorten <=1.3mm},
  lens/.style={draw,gray!55,thin},
]

% ---------- the network (left) ----------
\begin{scope}[on background layer]
  \draw[fill=gray!12,draw=gray!80,line width=0.8pt]
    plot[smooth cycle] coordinates {
      (0.00,2.05) (1.05,1.60) (1.55,0.70) (1.95,-0.15) (1.50,-1.15)
      (0.60,-1.85) (-0.55,-1.60) (-1.50,-1.85) (-1.95,-0.85) (-1.60,0.15)
      (-1.85,1.15) (-0.90,1.75)};
\end{scope}

\node[dev] (n1) at (-0.15, 1.25) {};
\node[dev] (n2) at (-1.05, 0.75) {};
\node[dev] (n3) at ( 0.85, 0.75) {};
\node[dev] (n4) at (-0.35, 0.55) {};
\node[dev] (n6) at (-0.85,-0.35) {};
\node[dev] (n7) at ( 0.35,-0.75) {};
\node[dev] (n8) at (-1.15,-1.15) {};
\node[dev] (n9) at ( 0.35,-1.45) {};
\node[dev,line width=0.7pt] (n5) at ( 1.35,-0.15) {};

\begin{scope}[on background layer]
  \foreach \a/\b in {n1/n2, n1/n3, n1/n4, n2/n4, n3/n4, n3/n5, n4/n6, n4/n7,
                     n5/n7, n6/n7, n6/n8, n7/n8, n7/n9, n8/n9}
    \draw[link] (\a) -- (\b);
\end{scope}

% ---------- zoom on a single device ----------
\draw[lens] (n5) -- (3.75, 1.40);
\draw[lens] (n5) -- (3.75,-1.40);

% ---------- the round (right) ----------
\draw[->,line width=0.8pt,gray!80] (3.85,1.70) -- (11.10,1.70)
  node[midway,above,text=black,font=\footnotesize] {computational round};

% neighbourhood of the device: offsets shared by the three snapshots
\def\nbrs{1/0.22/0.85, 2/-0.60/0.50, 3/0.85/0.12, 4/-0.58/-0.52, 5/0.44/-0.80}

% -- sense: the device gathers the values of its neighbours --
\def\px{4.75}
\node[hub] (sense) at (\px,0) {};
\foreach \i/\dx/\dy in \nbrs {
  \node[dev] (s\i) at ({\px+\dx},\dy) {};
  \draw[msg] (s\i) -- (sense);
}
\node[font=\small\bfseries] at (\px,-1.45) {Sense};

% -- compute: the device evaluates the program locally --
\def\px{7.40}
\node[hub] (comp) at (\px,0) {};
\foreach \i/\dx/\dy in \nbrs {
  \node[idle] (c\i) at ({\px+\dx},\dy) {};
  \draw[faint] (c\i) -- (comp);
}
\draw[->,gray!85] ([shift=(55:3.3mm)]comp) arc (55:-275:3.3mm);
\node[font=\small\bfseries] at (\px,-1.45) {Compute};

% -- interact: the device shares the result with its neighbours --
\def\px{10.05}
\node[hub] (int) at (\px,0) {};
\foreach \i/\dx/\dy in \nbrs {
  \node[dev] (i\i) at ({\px+\dx},\dy) {};
  \draw[msg] (int) -- (i\i);
}
\node[font=\small\bfseries] at (\px,-1.45) {Interact};

\end{tikzpicture}
