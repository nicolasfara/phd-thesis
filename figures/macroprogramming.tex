% Macroprogramming: a macro-program written against macro-abstractions is mapped
% to the micro-programs run by single entities, whose effects emerge as
% macro-observables. Redraw of figures/macroprogramming.svg.
% Requires: \usepackage{tikz}
%   \usetikzlibrary{arrows.meta,positioning,fit,backgrounds,shapes.geometric}
% Include with a normal figure environment, e.g. \input{figures/macroprogramming}.
\begin{tikzpicture}[
  >=Latex, font=\small,
  prog/.style={rounded corners,draw,fill=gray!6,minimum width=25mm,minimum height=16mm,align=center},
  blk/.style={draw,fill=gray!18,inner sep=0pt},
  ent/.style={draw,fill=gray!35,inner sep=0pt},
  circ/.style={ent,circle,minimum size=4mm},
  tri/.style={ent,regular polygon,regular polygon sides=3,minimum size=4.6mm},
  dia/.style={ent,diamond,minimum size=4.6mm},
  flow/.style={->,gray!80},
  wire/.style={->,dotted,gray!85,shorten >=0.8mm},
  lbl/.style={font=\footnotesize},
  hollow/.style={draw=gray!80,fill=gray!10},
]

% ---------- macro level ----------
\node[prog] (macro) at (2.15,5.45) {Macro\\program};

% macro-abstractions: three overlapping blocks
\draw[blk] (5.27,5.60) rectangle (6.24,6.03);
\draw[blk] (5.66,5.22) rectangle (6.51,5.66);
\draw[blk] (6.29,5.60) rectangle (7.12,6.04);
\node[lbl] at (6.20,6.45) {Macro-abstractions};
\draw[flow] (5.22,5.70) -- (macro.east);

% macro-observables: stacked wavy sheets
\foreach \b in {7.30,6.96,6.62}{
  \draw[hollow]
    plot[domain=0:1.45,samples=26,variable=\x] ({8.15+\x},{\b+0.09*sin(250*\x)})
    -- ({9.60},{\b-0.24+0.09*sin(250*1.45)})
    --plot[domain=1.45:0,samples=26,variable=\x] ({8.15+\x},{\b-0.24+0.09*sin(250*\x)})
    -- cycle;
}
\node[lbl] at (8.85,7.85) {Macro-observables};

% goals: from the macro program to what the system should be observed to do
\draw[flow] (macro.north east) to[out=55,in=175] (8.05,7.05);
\node[lbl] at (5.60,7.60) {Goals};

% ---------- level separator ----------
\draw[dotted,thick,gray!70] (-0.15,3.75) -- (13.45,3.75);
\node[font=\small\bfseries] at (12.10,5.65) {Macro-level};
\node[font=\small\bfseries] at (12.10,1.75) {Micro-level};

% ---------- the two vertical block arrows across the levels ----------
% abstractions (bottom-up, still to be built)
\draw[hollow,dashed]
  (6.01,3.10) -- (6.19,3.10) -- (6.19,4.72) -- (6.36,4.72)
  -- (6.10,5.08) -- (5.84,4.72) -- (6.01,4.72) -- cycle;
\node[lbl,anchor=east] at (5.75,4.05) {Abstractions};
% effects and emergence (bottom-up, observed)
\draw[hollow]
  (8.66,3.10) -- (8.84,3.10) -- (8.84,5.80) -- (9.03,5.80)
  -- (8.75,6.18) -- (8.47,5.80) -- (8.66,5.80) -- cycle;
\node[lbl,anchor=west] at (9.30,4.05) {Effects/emergence};

% ---------- micro level ----------
\node[prog] (micro) at (2.15,1.70) {Micro\\program(s)};
\draw[flow] (macro.south) -- (micro.north);
\node[lbl,anchor=west] at (2.35,3.25) {macro-to-micro};

% environment
\draw[draw=gray!80,fill=gray!6] (5.05,2.95) -- (9.40,2.95) -- (8.95,0.40) -- (4.60,0.40) -- cycle;

  \draw[wire] (3.42,\y) -- (\e);

\end{tikzpicture}
