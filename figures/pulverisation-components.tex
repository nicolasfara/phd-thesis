% Original five-role decomposition, following Casadei et al. (2020).
% Lines denote dependencies, not an execution order or physical placement.
\begin{tikzpicture}[
  font=\small,
  role/.style={draw,rounded corners,fill=gray!8,
    minimum width=27mm,minimum height=9mm,align=center},
  dependency/.style={thick,gray!75},
  neighbour/.style={draw,rounded corners,dashed,fill=gray!4,
    text width=23mm,minimum height=13mm,align=center,font=\scriptsize},
]
  \node[role] (behaviour) at (0,0) {Behaviour};
  \node[role] (state) at (-1.9,1.4) {State};
  \node[role] (communication) at (1.9,1.4) {Communication};
  \node[role] (sensing) at (-1.9,-1.4) {Sensing};
  \node[role] (actuation) at (1.9,-1.4) {Actuation};

  \draw[dependency] (behaviour) -- (state);
  \draw[dependency] (behaviour) -- (communication);
  \draw[dependency] (behaviour) -- (sensing);
  \draw[dependency] (behaviour) -- (actuation);

  \begin{scope}[on background layer]
    \node[draw,rounded corners,fill=gray!3,
      fit=(state)(communication)(sensing)(actuation),inner sep=5mm] (logical) {};
  \end{scope}
  \node[anchor=south] at ([yshift=1mm]logical.north) {Logical device};

  \node[neighbour] (n1) at (6.1,1.4) {Neighbouring\\logical device};
  \node[neighbour] (n2) at (6.1,-1.1) {Neighbouring\\logical device};
  \draw[dependency] (communication.east) -- (n1.west);
  \draw[dependency] (communication.east) -- ++(0.8,0) |- (n2.west);
\end{tikzpicture}
