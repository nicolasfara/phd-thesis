% Original five-role decomposition, following Casadei et al. (2020).
% Lines denote dependencies, not an execution order or physical placement.
% Colours and link styles match pulverisation-deployments.tex: dotted boxes are
% logical elements, one colour per logical device.
\colorlet{pdA}{blue!18}
\colorlet{pdB}{red!18}
\colorlet{pdC}{green!25}
\begin{tikzpicture}[
  font=\small,
  role/.style={draw,densely dotted,fill=pdA,
    minimum width=30mm,minimum height=9mm,align=center},
  dependency/.style={thick,gray!75},
  neighbourlink/.style={red!75!black,densely dotted,line width=1pt},
  neighbour/.style={draw,densely dotted,fill=#1,
    text width=23mm,minimum height=13mm,align=center,font=\scriptsize},
]
  \node[role] (behaviour) at (0,0) {Behaviour ($\beta$)};
  \node[role] (state) at (-2.0,1.4) {State ($\kappa$)};
  \node[role] (communication) at (2.0,1.4) {Communication ($\chi$)};
  \node[role] (sensing) at (-2.0,-1.4) {Sensing ($\sigma$)};
  \node[role] (actuation) at (2.0,-1.4) {Actuation ($\alpha$)};

  \draw[dependency] (behaviour) -- (state);
  \draw[dependency] (behaviour) -- (communication);
  \draw[dependency] (behaviour) -- (sensing);
  \draw[dependency] (behaviour) -- (actuation);

  \begin{scope}[on background layer]
    \node[draw,densely dotted,rounded corners,fill=pdA!25,
      fit=(state)(communication)(sensing)(actuation),inner sep=5mm] (logical) {};
  \end{scope}
  \node[anchor=south] at ([yshift=1mm]logical.north) {Logical device};

  \node[neighbour=pdB] (n1) at (6.3,1.4) {Neighbouring\\logical device};
  \node[neighbour=pdC] (n2) at (6.3,-1.1) {Neighbouring\\logical device};
  \draw[neighbourlink] (communication.east) -- (n1.west);
  \draw[neighbourlink] (communication.east) -- ++(0.8,0) |- (n2.west);
\end{tikzpicture}
